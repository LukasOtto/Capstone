from flask import Flask, request, jsonify
from flask_cors import CORS
from google.cloud import bigquery
import datetime

app = Flask(__name__)
CORS(app)

client = bigquery.Client()


PROJECT_ID = "gear-management-472216"
DATASET_ID = "productos"
TABLA_BASE = f"{PROJECT_ID}.{DATASET_ID}.productos"


@app.route("/crear_tabla", methods=["POST"])
def crear_tabla():
    try:
        data = request.get_json()
        fecha = data.get("fecha")

        if not fecha:
           
            fecha = datetime.date.today().strftime("%Y_%m_%d")

        nombre_tabla = f"{DATASET_ID}.{fecha}"
        table_id = f"{PROJECT_ID}.{nombre_tabla}"

        
        tablas = list(client.list_tables(DATASET_ID))
        if any(t.table_id == fecha for t in tablas):
            return jsonify({"message": f"La tabla '{fecha}' ya existe."}), 200

        
        query = f"""
            CREATE TABLE `{table_id}` AS
            SELECT sku, nombre, categoria, ubicacion, 0 AS cantidad_escaneos
            FROM `{TABLA_BASE}`;
        """
        client.query(query).result()

        return jsonify({"message": f"Tabla '{fecha}' creada exitosamente."}), 201

    except Exception as e:
        print("Error creando tabla:", e)
        return jsonify({"error": str(e)}), 500
